class Solution:
    def searchRange(self, nums: List[int], target: int) -> List[int]:
        return [self.findFirst(nums, target), self.findLast(nums, target)]
    
    def findFirst(self, nums, target):
        low, high = 0, len(nums) - 1
        ans = -1
        
        while low <= high:
            mid = (low + high) // 2
            
            if nums[mid] == target:
                ans = mid
                high = mid - 1
            elif nums[mid] < target:
                low = mid + 1
            else:
                high = mid - 1
        
        return ans
    
    def findLast(self, nums, target):
        low, high = 0, len(nums) - 1
        ans = -1
        
        while low <= high:
            mid = (low + high) // 2
            
            if nums[mid] == target:
                ans = mid
                low = mid + 1
            elif nums[mid] < target:
                low = mid + 1
            else:
                high = mid - 1
        
        return ans
